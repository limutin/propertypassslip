┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃  ① GET YOUR FIREBASE SETTINGS                              ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

   1. Open this website:
      https://console.firebase.google.com/

   2. Click on YOUR project name

   3. Click the GEAR icon ⚙️ on the left side

   4. Click "Project Settings"

   5. Scroll DOWN until you see "Your apps"

   6. Do you see a "</>" icon with a name under it?
      
      YES → Click on it
      NO  → Click the "</>" button to add a web app
            Give it any name (like "PropertyPassSlip")
            Click "Register app"

   7. You'll see code that starts with:
      const firebaseConfig = {

   8. COPY everything from { to } including the { }

   9. Keep this copied! You'll need it in the next step.

   ✅ DONE? → Go to step ②


┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃  ② UPDATE YOUR CODE FILE                                   ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

   1. Open your code editor (VS Code, Notepad++, etc.)

   2. Open this file:
      src/firebase/config.js

   3. You'll see this part:
      const firebaseConfig = {
        apiKey: "YOUR_API_KEY",
        authDomain: "YOUR_AUTH_DOMAIN",
        ...
      };

   4. SELECT and DELETE everything between the { and }
      (from line 8 to line 14, including those lines)

   5. PASTE what you copied in step ①

   6. Press CTRL + S to SAVE

   7. Close the file

   ✅ DONE? → Go to step ③


┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃  ③ LOGIN TO FIREBASE                                       ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

   1. Find this file in your project folder:
      firebase-login.bat

   2. DOUBLE-CLICK on it

   3. A window will open, then your web browser will open

   4. SIGN IN with your Google account
      (Use the same account you used for Firebase)

   5. You'll see "Success!" or "Login successful"

   6. Close the browser tab

   7. The command window will show you're logged in

   ✅ DONE? → Go to step ④


┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃  ④ CONNECT TO YOUR PROJECT                                 ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

   1. Find this file:
      firebase-init.bat

   2. DOUBLE-CLICK on it

   3. A window will appear asking questions. Answer like this:

   ┌────────────────────────────────────────────────────────┐
   │ Question: Which Firebase features?                     │
   │ Answer: Press SPACEBAR next to "Hosting"              │
   │         Then press ENTER                               │
   └────────────────────────────────────────────────────────┘

   ┌────────────────────────────────────────────────────────┐
   │ Question: Please select an option                      │
   │ Answer: Choose "Use an existing project"              │
   │         Press ENTER                                    │
   └────────────────────────────────────────────────────────┘

   ┌────────────────────────────────────────────────────────┐
   │ Question: Select a default Firebase project            │
   │ Answer: Choose YOUR project name from the list         │
   │         Press ENTER                                    │
   └────────────────────────────────────────────────────────┘

   ┌────────────────────────────────────────────────────────┐
   │ Question: What do you want as public directory?        │
   │ Answer: Type: dist                                     │
   │         Press ENTER                                    │
   └────────────────────────────────────────────────────────┘

   ┌────────────────────────────────────────────────────────┐
   │ Question: Configure as single-page app?                │
   │ Answer: Type: Y                                        │
   │         Press ENTER                                    │
   └────────────────────────────────────────────────────────┘

   ┌────────────────────────────────────────────────────────┐
   │ Question: Set up automatic builds with GitHub?         │
   │ Answer: Type: N                                        │
   │         Press ENTER                                    │
   └────────────────────────────────────────────────────────┘

   4. You'll see "Firebase initialization complete!"

   ✅ DONE? → Go to step ⑤


┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃  ⑤ DEPLOY YOUR WEBSITE                                     ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

   1. Find this file:
      deploy.bat

   2. DOUBLE-CLICK on it

   3. WAIT... This takes 3-5 minutes
      You'll see:
      • "Installing dependencies..." (takes longest)
      • "Building project..."
      • "Deploying to Firebase..."

   4. When it's done, you'll see:
      ✔ Deploy complete!
      
      Hosting URL: https://something.web.app

   5. COPY that URL! That's your live website!

   ✅ DONE? → Go to step ⑥

IF SOMETHING GOES WRONG:
════════════════════════

❌ "npm not found" or "firebase not found"
   → Open Command Prompt
   → Type: npm install -g firebase-tools
   → Press Enter, wait
   → Try again from step ③

❌ Window closes immediately
   → Right-click the .bat file
   → Choose "Edit"
   → Add "pause" at the end
   → Save and try again

❌ "Not authorized"
   → Go back to step ③ (firebase-login.bat)

❌ Website shows blank page
   → Did you do step ② correctly?
   → Check src/firebase/config.js
   → Make sure it has real values, not "YOUR_API_KEY"

❌ Still stuck?
   → Open SIMPLE_STEPS.md for more details
   → Or see VISUAL_GUIDE.txt
